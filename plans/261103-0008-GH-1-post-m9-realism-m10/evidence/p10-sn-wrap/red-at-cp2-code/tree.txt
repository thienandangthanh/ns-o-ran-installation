// Red-run stand-in: reproduces the unfixed behaviour exactly, so the suite can be shown to
// fail against it. Not part of any commit.
//  - OrderPdcpSnsByCount: the std::map<uint32_t, Ptr<Packet>> keyed by PDCP SN in
//    LteRlcAm::CreateRlcSduBuffer iterates in ascending raw SN order.
//  - IsPdcpSnInForwardingLookback: the eNB/UE RRC test
//    `sn >= (SN_1 - 2) && sn <= SN_1` on uint16_t operands (promoted to int).

#include "lte-pdcp-sn-wrap.h"

#include <algorithm>

namespace ns3
{

std::vector<uint16_t>
OrderPdcpSnsByCount(std::vector<uint16_t> sns)
{
    std::sort(sns.begin(), sns.end());
    return sns;
}

bool
IsPdcpSnInForwardingLookback(uint16_t sn, uint16_t firstForwardedSn)
{
    return sn >= (firstForwardedSn - 2) && sn <= (firstForwardedSn);
}

} // namespace ns3
